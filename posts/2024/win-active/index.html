     1#requires -runasadministrator
 2[cmdletbinding()]
 3param(
 4    [Parameter(Position = 1, Mandatory = $true, HelpMessage='set WSL IP,default port 1688,Run as Admin')]

 6    $ip
 7)
 8# 删除激活信息
 9# slmgr.vbs -upk; 
10# 专业工作站 key
11# slmgr.vbs -ipk NRG8B-VKK3Q-CXVCJ-9G2XF-6Q84J;
